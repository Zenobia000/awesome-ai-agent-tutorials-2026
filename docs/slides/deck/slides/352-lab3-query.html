from sentence_transformers import CrossEncoder def rrf(rankings, k=60): score = {} for ranking in rankings: for i, d in enumerate(ranking, 1): score[d] = score.get(d, 0) + 1 / (k + i) return sorted(score, key=score.get, reverse=True) fused = rrf([sparse, dense])[:20] reranker = CrossEncoder("BAAI/bge-reranker-v2-m3") scores = reranker.predict([(q, text[d]) for d in fused]) ranked = sorted(zip(scores, fused), reverse=True) prompt = "只根據以下資料回答：\n" + "\n---\n".join( text[d] for _, d in ranked[:3])
split() 切不開中文，所以取相鄰兩字（bigram）；0 分＝一個詞都沒對上，不進榜；BM25 也套同一個 version 過濾。
